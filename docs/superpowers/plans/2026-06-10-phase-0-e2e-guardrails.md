# Phase 0 E2E Guardrails Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add repeatable E2E guardrails for the current React prototype so later API, provider, and Mastra work can proceed without breaking the existing product flow.

**Architecture:** Keep Vitest for domain/unit coverage and add Playwright for browser-level product workflow checks. Playwright starts the existing Vite dev server, tests the 10 primary navigation views at desktop and mobile widths, and verifies the current local-demo behavior for search, corpus learning, podcast generation, and artifact deletion.

**Tech Stack:** React 18, Vite 7, TypeScript 5, Vitest, Playwright Test.

import { expect, type Page } from "@playwright/test";

export async function resetDemoState(page: Page) {
  await page.goto("/");
  await page.evaluate(() => window.localStorage.clear());
  await page.reload();
  await expect(page.getByRole("button", { name: "今日访谈台" })).toBeVisible();
}

export async function openView(page: Page, navName: string, expectedText: string) {
  await page.getByRole("button", { name: navName }).click();
  await expect(page.getByText(expectedText)).toBeVisible();
}

export async function expectNoHorizontalOverflow(page: Page) {
  const dimensions = await page.evaluate(() => ({
    clientWidth: document.documentElement.clientWidth,
    scrollWidth: document.documentElement.scrollWidth
  }));
  expect(dimensions.scrollWidth).toBeLessThanOrEqual(dimensions.clientWidth);
}

export async function useMobileViewport(page: Page) {
  await page.setViewportSize({ width: 390, height: 844 });
}
```

import { test } from "@playwright/test";
import { expectNoHorizontalOverflow, openView, resetDemoState, useMobileViewport } from "./helpers";

const views = [
  ["今日访谈台", "你对“救火”和放手的真实边界"],
  ["实时专访", "专访启动"],
  ["联网素材", "主动搜集热点、素材和可讨论主题"],
  ["语料学习", "从外部专访中学习提问逻辑"],
  ["播客工作台", "把一次专访生成可收听的语音节目"],
  ["双轨笔记", "从访谈沉淀认知资产"],
  ["资产库", "观点、问题、决策和知识债"],
  ["长期追踪", "看见反复主题和观点变化"],
  ["系统配置", "底层能力、provider 和引擎策略"],
  ["个人档案", "产品负责人 · AI 产品 / 创业"]
] as const;

test.describe("primary navigation", () => {
  test("opens every primary view on desktop without horizontal overflow", async ({ page }) => {
    await resetDemoState(page);

    for (const [navName, expectedText] of views) {
      await openView(page, navName, expectedText);
      await expectNoHorizontalOverflow(page);
    }
  });

  test("opens every primary view on mobile without horizontal overflow", async ({ page }) => {
    await useMobileViewport(page);
    await resetDemoState(page);

    for (const [navName, expectedText] of views) {
      await openView(page, navName, expectedText);
      await expectNoHorizontalOverflow(page);
    }
  });
});
```

import { expectNoHorizontalOverflow, openView, resetDemoState } from "./helpers";
